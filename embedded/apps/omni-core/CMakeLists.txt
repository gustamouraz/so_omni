cmake_minimum_required(VERSION 3.16)
project(omni_core VERSION 0.2.1 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  # The production target has 1 GB of RAM and 16 GB of flash.  Prefer a small
  # executable and keep symbols only in development builds.
  add_compile_options($<$<CONFIG:Release>:-Os> $<$<CONFIG:Release>:-DNDEBUG>)
  add_link_options($<$<CONFIG:Release>:-Wl,--as-needed>)
endif()
add_library(omni-state STATIC src/state.c)
target_include_directories(omni-state PUBLIC include)
target_compile_options(omni-state PRIVATE -Wall -Wextra -Wpedantic)
add_executable(omni-core src/main.c)
target_link_libraries(omni-core PRIVATE omni-state)
install(TARGETS omni-core RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
  add_executable(omni-state-test test/test_state.c)
  target_link_libraries(omni-state-test PRIVATE omni-state)
  add_test(NAME omni-state-test COMMAND omni-state-test)
endif()
